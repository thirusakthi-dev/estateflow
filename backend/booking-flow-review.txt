import { Prisma, BookingStatus } from "../../generated/prisma/client";
import { prisma } from "../../lib/prisma";
import { HTTP_STATUS } from "../../constants/http-status";
import { ApiError } from "../../utils/api-error";

import {
  createBookingSchema,
  listBookingsSchema,
  updateBookingStatusSchema,
} from "./booking.validator";

const bookingSelect = {
  id: true,
  leadId: true,
  buildingId: true,
  bookedById: true,
  status: true,
  bookedAt: true,
  cancelledAt: true,

  lead: {
    select: {
      id: true,
      name: true,
      phone: true,
      assignedToId: true,
    },
  },

  building: {
    select: {
      id: true,
      name: true,
      project: {
        select: {
          id: true,
          name: true,
          city: true,
        },
      },
    },
  },

  bookedBy: {
    select: {
      id: true,
      name: true,
      email: true,
    },
  },
} satisfies Prisma.BookingSelect;

type BookingUser = {
  id: string;
  role: string;
};

function canAccessBooking(
  user: BookingUser,
  booking: {
    bookedById: string;
    lead: {
      assignedToId: string | null;
    };
  },
) {
  return (
    user.role === "ADMIN" ||
    booking.bookedById === user.id ||
    booking.lead.assignedToId === user.id
  );
}

function hasAccessToLead(user: BookingUser, assignedToId: string | null) {
  return user.role === "ADMIN" || assignedToId === user.id;
}

/**
 * Create a pending booking.
 */
export async function createBooking(input: unknown, user: BookingUser) {
  const parsed = createBookingSchema.safeParse(input);

  if (!parsed.success) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Invalid booking data.",
      parsed.error.flatten(),
    );
  }

  const { leadId, buildingId } = parsed.data;

  return prisma.$transaction(async (tx) => {
    const lead = await tx.lead.findUnique({
      where: { id: leadId },
      select: {
        id: true,
        assignedToId: true,
        stage: true,
      },
    });

    if (!lead) {
      throw new ApiError(HTTP_STATUS.NOT_FOUND, "Lead not found.");
    }

    if (!hasAccessToLead(user, lead.assignedToId)) {
      throw new ApiError(HTTP_STATUS.FORBIDDEN, "You cannot book this lead.");
    }

    if (lead.stage === "LOST") {
      throw new ApiError(
        HTTP_STATUS.BAD_REQUEST,
        "A lost lead cannot be booked.",
      );
    }

    if (lead.stage === "BOOKED") {
      throw new ApiError(
        HTTP_STATUS.CONFLICT,
        "This lead already has a booking.",
      );
    }

    const building = await tx.building.findUnique({
      where: { id: buildingId },
      select: { id: true },
    });

    if (!building) {
      throw new ApiError(HTTP_STATUS.NOT_FOUND, "Building not found.");
    }

    const booking = await tx.booking.create({
      data: {
        leadId,
        buildingId,
        bookedById: user.id,
        status: BookingStatus.PENDING,
      },
      select: bookingSelect,
    });

    // The lead becomes BOOKED only after the booking is confirmed.
    return booking;
  });
}

/**
 * List bookings with filters and pagination.
 */
export async function listBookings(user: BookingUser, input: unknown) {
  const parsed = listBookingsSchema.safeParse(input);

  if (!parsed.success) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Invalid booking filters.",
      parsed.error.flatten(),
    );
  }

  const {
    page = 1,
    limit = 10,
    search,
    status,
    leadId,
    buildingId,
    from,
    to,
  } = parsed.data;

  const conditions: Prisma.BookingWhereInput[] = [];

  if (user.role !== "ADMIN") {
    conditions.push({
      OR: [{ bookedById: user.id }, { lead: { assignedToId: user.id } }],
    });
  }

  if (status) {
    conditions.push({ status });
  }

  if (leadId) {
    conditions.push({ leadId });
  }

  if (buildingId) {
    conditions.push({ buildingId });
  }

  if (from || to) {
    conditions.push({
      bookedAt: {
        ...(from ? { gte: new Date(from) } : {}),
        ...(to ? { lte: new Date(to) } : {}),
      },
    });
  }

  if (search) {
    conditions.push({
      OR: [
        {
          lead: {
            name: {
              contains: search,
              mode: "insensitive",
            },
          },
        },
        {
          building: {
            name: {
              contains: search,
              mode: "insensitive",
            },
          },
        },
        {
          building: {
            project: {
              name: {
                contains: search,
                mode: "insensitive",
              },
            },
          },
        },
      ],
    });
  }

  const where: Prisma.BookingWhereInput = {
    AND: conditions,
  };

  const [bookings, total] = await prisma.$transaction([
    prisma.booking.findMany({
      where,
      select: bookingSelect,
      orderBy: { bookedAt: "desc" },
      skip: (page - 1) * limit,
      take: limit,
    }),

    prisma.booking.count({ where }),
  ]);

  return {
    bookings,
    pagination: {
      page,
      limit,
      total,
      totalPages: Math.ceil(total / limit),
    },
  };
}

/**
 * Get one booking.
 */
export async function getBookingById(id: string, user: BookingUser) {
  const booking = await prisma.booking.findUnique({
    where: { id },
    select: bookingSelect,
  });

  if (!booking) {
    throw new ApiError(HTTP_STATUS.NOT_FOUND, "Booking not found.");
  }

  if (!canAccessBooking(user, booking)) {
    throw new ApiError(
      HTTP_STATUS.FORBIDDEN,
      "You cannot access this booking.",
    );
  }

  return booking;
}

/**
 * Update booking status.
 *
 * PENDING   -> CONFIRMED or CANCELLED
 * CONFIRMED -> CANCELLED
 * CANCELLED -> CONFIRMED
 */
export async function updateBookingStatus(
  id: string,
  input: unknown,
  user: BookingUser,
) {
  const parsed = updateBookingStatusSchema.safeParse(input);

  if (!parsed.success) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Invalid booking status.",
      parsed.error.flatten(),
    );
  }

  const { status } = parsed.data;

  return prisma.$transaction(async (tx) => {
    const existing = await tx.booking.findUnique({
      where: { id },
      select: {
        id: true,
        leadId: true,
        bookedById: true,
        status: true,

        lead: {
          select: {
            id: true,
            stage: true,
            assignedToId: true,
          },
        },
      },
    });

    if (!existing) {
      throw new ApiError(HTTP_STATUS.NOT_FOUND, "Booking not found.");
    }

    if (!canAccessBooking(user, existing)) {
      throw new ApiError(
        HTTP_STATUS.FORBIDDEN,
        "You cannot update this booking.",
      );
    }

    if (existing.status === status) {
      return tx.booking.findUniqueOrThrow({
        where: { id },
        select: bookingSelect,
      });
    }

    // Cancel a pending or confirmed booking.
    if (status === BookingStatus.CANCELLED) {
      if (existing.status === BookingStatus.CANCELLED) {
        return tx.booking.findUniqueOrThrow({
          where: { id },
          select: bookingSelect,
        });
      }

      const result = await tx.booking.updateMany({
        where: {
          id,
          status: existing.status,
        },
        data: {
          status: BookingStatus.CANCELLED,
          cancelledAt: new Date(),
        },
      });

      if (result.count === 0) {
        throw new ApiError(
          HTTP_STATUS.CONFLICT,
          "Booking status has changed. Refresh and try again.",
        );
      }

      // Only a previously confirmed booking should restore the lead.
      if (existing.status === BookingStatus.CONFIRMED) {
        await tx.lead.updateMany({
          where: {
            id: existing.leadId,
            stage: "BOOKED",
          },
          data: {
            stage: "INTERESTED",
          },
        });
      }
    }

    // Confirm a pending or previously cancelled booking.
    if (status === BookingStatus.CONFIRMED) {
      const lead = existing.lead;

      if (lead.stage === "LOST") {
        throw new ApiError(
          HTTP_STATUS.BAD_REQUEST,
          "A lost lead cannot be booked.",
        );
      }

      if (lead.stage === "BOOKED") {
        throw new ApiError(
          HTTP_STATUS.CONFLICT,
          "This lead already has an active booking.",
        );
      }

      const otherConfirmedBooking = await tx.booking.findFirst({
        where: {
          leadId: existing.leadId,
          id: { not: id },
          status: BookingStatus.CONFIRMED,
        },
        select: { id: true },
      });

      if (otherConfirmedBooking) {
        throw new ApiError(
          HTTP_STATUS.CONFLICT,
          "This lead already has another confirmed booking.",
        );
      }

      const result = await tx.booking.updateMany({
        where: {
          id,
          status: existing.status,
        },
        data: {
          status: BookingStatus.CONFIRMED,
          cancelledAt: null,
        },
      });

      if (result.count === 0) {
        throw new ApiError(
          HTTP_STATUS.CONFLICT,
          "Booking status has changed. Refresh and try again.",
        );
      }

      await tx.lead.update({
        where: { id: existing.leadId },
        data: { stage: "BOOKED" },
      });
    }

    return tx.booking.findUniqueOrThrow({
      where: { id },
      select: bookingSelect,
    });
  });
}

export async function cancelBooking(id: string, user: BookingUser) {
  return updateBookingStatus(id, { status: BookingStatus.CANCELLED }, user);
}

import { z } from "zod";

export const createBookingSchema = z.object({
  leadId: z.string().trim().min(1, "Lead ID is required."),
  buildingId: z.string().trim().min(1, "Building ID is required."),
});

export const listBookingsSchema = z
  .object({
    search: z.string().trim().max(100).optional(),
    status: z.enum(["PENDING", "CONFIRMED", "CANCELLED"]).optional(),
    leadId: z.string().trim().min(1).optional(),
    buildingId: z.string().trim().min(1).optional(),
    from: z.string().datetime({ offset: true }).optional(),
    to: z.string().datetime({ offset: true }).optional(),
    page: z.coerce.number().int().min(1).default(1),
    limit: z.coerce.number().int().min(1).max(100).default(10),
  })
  .refine(
    (data) =>
      !data.from ||
      !data.to ||
      new Date(data.from).getTime() <= new Date(data.to).getTime(),
    {
      message: "from cannot be later than to.",
      path: ["from"],
    },
  );

export const updateBookingStatusSchema = z.object({
  status: z.enum(["PENDING", "CONFIRMED", "CANCELLED"]),
});

export async function createBuilding(
  input: unknown,
  files: Express.Multer.File[] = [],
) {
  const parsed = createBuildingSchema.safeParse(input);

  if (!parsed.success) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Invalid building details.",
      parsed.error.flatten(),
    );
  }

  if (files.length > 10) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Upload a maximum of 10 images per request.",
    );
  }

  const project = await prisma.project.findUnique({
    where: { id: parsed.data.projectId },
    select: { id: true, status: true },
  });

  if (!project) {
    throw new ApiError(HTTP_STATUS.NOT_FOUND, "Project not found.");
  }

  if (project.status !== "ACTIVE") {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Cannot add a building to an inactive project.",
    );
  }

  const uploaded: { imageUrl: string; publicId: string }[] = [];

  try {
    for (const file of files) {
      uploaded.push(await uploadImage(file, "buildings"));
    }

    return await prisma.$transaction((tx) =>
      tx.building.create({
        data: {
          ...parsed.data,
          images: {
            create: uploaded.map((image, index) => ({
              imageUrl: image.imageUrl,
              publicId: image.publicId,
              isCover: index === 0,
              sortOrder: index,
            })),
          },
        },
        select: buildingSelect,
      }),
    );
  } catch (error) {
    await Promise.allSettled(
      uploaded.map((image) => deleteImage(image.publicId)),
    );

    throw error;
  }
}

export async function updateBuilding(
  id: string,
  input: unknown,
  files: Express.Multer.File[] = [],
) {
  const body: Record<string, unknown> =
    typeof input === "object" && input !== null && !Array.isArray(input)
      ? { ...(input as Record<string, unknown>) }
      : {};

  let removedImageIds: string[] = [];

  try {
    const rawRemovedIds = body.removedImageIds;
    delete body.removedImageIds;

    if (typeof rawRemovedIds === "string") {
      const parsedIds: unknown = JSON.parse(rawRemovedIds);

      if (
        !Array.isArray(parsedIds) ||
        !parsedIds.every((value) => typeof value === "string")
      ) {
        throw new Error("Invalid removed image IDs");
      }

      removedImageIds = parsedIds;
    } else if (Array.isArray(rawRemovedIds)) {
      if (!rawRemovedIds.every((value) => typeof value === "string")) {
        throw new Error("Invalid removed image IDs");
      }

      removedImageIds = rawRemovedIds;
    } else if (rawRemovedIds !== undefined) {
      throw new Error("Invalid removed image IDs");
    }
  } catch {
    throw new ApiError(HTTP_STATUS.BAD_REQUEST, "Invalid removed image IDs.");
  }

  const parsed = updateBuildingSchema.safeParse(body);

  if (!parsed.success) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Invalid building details.",
      parsed.error.flatten(),
    );
  }

  if (files.length > 10) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Upload a maximum of 10 images per request.",
    );
  }

  const existing = await prisma.building.findUnique({
    where: { id },
    select: {
      id: true,
      images: {
        select: {
          id: true,
          publicId: true,
        },
      },
    },
  });

  if (!existing) {
    throw new ApiError(HTTP_STATUS.NOT_FOUND, "Building not found.");
  }

  const uniqueRemovedIds = [...new Set(removedImageIds)];

  const removedImages = existing.images.filter((image) =>
    uniqueRemovedIds.includes(image.id),
  );

  if (removedImages.length !== uniqueRemovedIds.length) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "One or more images do not belong to this building.",
    );
  }

  if (
    Object.keys(parsed.data).length === 0 &&
    files.length === 0 &&
    removedImages.length === 0
  ) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Provide at least one field or image change to update.",
    );
  }

  const uploaded: { imageUrl: string; publicId: string }[] = [];

  try {
    for (const file of files) {
      uploaded.push(await uploadImage(file, "buildings"));
    }

    const building = await prisma.$transaction(async (tx) => {
      if (removedImages.length > 0) {
        await tx.buildingImage.deleteMany({
          where: {
            id: { in: removedImages.map((image) => image.id) },
            buildingId: id,
          },
        });
      }

      if (uploaded.length > 0) {
        const remainingCount = existing.images.length - removedImages.length;

        await tx.buildingImage.createMany({
          data: uploaded.map((image, index) => ({
            buildingId: id,
            imageUrl: image.imageUrl,
            publicId: image.publicId,
            isCover: false,
            sortOrder: remainingCount + index,
          })),
        });
      }

      // Keep exactly one cover image when images exist.
      const remainingImages = await tx.buildingImage.findMany({
        where: { buildingId: id },
        select: { id: true },
        orderBy: [{ sortOrder: "asc" }, { createdAt: "asc" }],
      });

      if (remainingImages.length > 0) {
        await tx.buildingImage.updateMany({
          where: { buildingId: id },
          data: { isCover: false },
        });

        await tx.buildingImage.update({
          where: { id: remainingImages[0].id },
          data: { isCover: true },
        });
      }

      return tx.building.update({
        where: { id },
        data: parsed.data,
        select: buildingSelect,
      });
    });

    // Remove deleted images from Cloudinary after the DB update succeeds.
    await Promise.allSettled(
      removedImages.map((image) => deleteImage(image.publicId)),
    );

    return building;
  } catch (error) {
    // Clean up newly uploaded Cloudinary files if the DB update fails.
    await Promise.allSettled(
      uploaded.map((image) => deleteImage(image.publicId)),
    );

    throw error;
  }
}

export async function deleteBuilding(id: string) {
  const building = await prisma.building.findUnique({
    where: { id },
    select: {
      id: true,
      units: {
        select: {
          id: true,
          status: true,
        },
      },
      images: {
        select: {
          publicId: true,
        },
      },
    },
  });

  if (!building) {
    throw new ApiError(HTTP_STATUS.NOT_FOUND, "Building not found.");
  }

  if (building.units.some((unit) => unit.status === "BOOKED")) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Cannot delete a building containing booked units.",
    );
  }

  await prisma.building.delete({
    where: { id },
  });

  await Promise.allSettled(
    building.images.map((image) => deleteImage(image.publicId)),
  );

  return { id: building.id };
}

export async function createFollowUp(user: CurrentUser, input: unknown) {
  const parsed = createFollowUpSchema.safeParse(input);

  if (!parsed.success) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Invalid follow-up data.",
      parsed.error.flatten(),
    );
  }

  const { leadId, assignedToId, scheduledAt, notes } = parsed.data;

  const lead = await prisma.lead.findUnique({
    where: { id: leadId },
    select: {
      id: true,
      assignedToId: true,
    },
  });

  if (!lead) {
    throw new ApiError(HTTP_STATUS.NOT_FOUND, "Lead not found.");
  }

  if (user.role !== "ADMIN" && lead.assignedToId !== user.id) {
    throw new ApiError(
      HTTP_STATUS.FORBIDDEN,
      "You can only schedule follow-ups for your assigned leads.",
    );
  }

  const targetUserId =
    user.role === "ADMIN" ? (assignedToId ?? lead.assignedToId) : user.id;

  if (!targetUserId) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Assign the lead to a sales employee before scheduling a follow-up.",
    );
  }

  const employee = await prisma.user.findUnique({
    where: { id: targetUserId },
    select: {
      id: true,
      role: true,
      isActive: true,
    },
  });

  if (!employee || employee.role !== "SALES_EMPLOYEE" || !employee.isActive) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Assigned employee is invalid or inactive.",
    );
  }

  return prisma.followUp.create({
    data: {
      leadId,
      assignedToId: targetUserId,
      scheduledAt,
      notes,
    },
    select: followUpSelect,
  });
}

export async function listFollowUps(user: CurrentUser, input: unknown) {
  const parsed = listFollowUpsSchema.safeParse(input);

  if (!parsed.success) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Invalid follow-up filters.",
      parsed.error.flatten(),
    );
  }

  const { status, leadId, from, to, page, limit } = parsed.data;

  const conditions: Prisma.FollowUpWhereInput[] = [];

  if (user.role !== "ADMIN") {
    conditions.push({
      OR: [{ assignedToId: user.id }, { lead: { assignedToId: user.id } }],
    });
  }

  if (status) {
    conditions.push({ status });
  }

  if (leadId) {
    conditions.push({ leadId });
  }

  if (from || to) {
    conditions.push({
      scheduledAt: {
        ...(from ? { gte: new Date(from) } : {}),
        ...(to ? { lte: new Date(to) } : {}),
      },
    });
  }

  const where: Prisma.FollowUpWhereInput = {
    AND: conditions,
  };

  const [followUps, total] = await Promise.all([
    prisma.followUp.findMany({
      where,
      skip: (page - 1) * limit,
      take: limit,
      orderBy: { scheduledAt: "asc" },
      select: followUpSelect,
    }),
    prisma.followUp.count({ where }),
  ]);

  return {
    followUps,
    pagination: {
      page,
      limit,
      total,
      totalPages: Math.ceil(total / limit),
    },
  };
}

export async function updateFollowUp(
  id: string,
  user: CurrentUser,
  input: unknown,
) {
  const parsed = updateFollowUpSchema.safeParse(input);

  if (!parsed.success) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Invalid follow-up update.",
      parsed.error.flatten(),
    );
  }

  const existing = await getAccessibleFollowUp(id, user);
  const { outcome, status, ...otherUpdates } = parsed.data;

  const outcomeProvided = outcome !== undefined;
  const outcomeChanged = outcomeProvided && outcome !== existing.outcome;
  const finalStatus = status ?? existing.status;

  if (outcomeProvided && finalStatus !== "COMPLETED") {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Complete the follow-up before recording its outcome.",
    );
  }

  return prisma.$transaction(async (tx) => {
    const followUp = await tx.followUp.update({
      where: { id },
      data: {
        ...otherUpdates,
        ...(status !== undefined ? { status } : {}),
        ...(outcomeProvided ? { outcome } : {}),
      },
      select: followUpSelect,
    });

    if (outcomeChanged && outcome) {
      const nextStage = outcomeToLeadStage[outcome];

      if (nextStage) {
        await tx.lead.updateMany({
          where: {
            id: existing.leadId,
            stage: { notIn: ["BOOKED", "LOST"] },
          },
          data: { stage: nextStage },
        });
      }
    }

    return followUp;
  });
}

export async function createProject(
  input: unknown,
  files: Express.Multer.File[] = [],
) {
  const parsed = createProjectSchema.safeParse(input);

  if (!parsed.success) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Invalid project details.",
      parsed.error.flatten(),
    );
  }

  if (files.length > 10) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Upload a maximum of 10 images per request.",
    );
  }

  const uploaded: { imageUrl: string; publicId: string }[] = [];

  try {
    for (const file of files) {
      uploaded.push(await uploadImage(file, "projects"));
    }

    return await prisma.$transaction((tx) =>
      tx.project.create({
        data: {
          ...parsed.data,
          images: {
            create: uploaded.map((image, index) => ({
              imageUrl: image.imageUrl,
              publicId: image.publicId,
              isCover: index === 0,
              sortOrder: index,
            })),
          },
        },
        select: projectSelect,
      }),
    );
  } catch (error) {
    await Promise.allSettled(
      uploaded.map((image) => deleteImage(image.publicId)),
    );

    throw error;
  }
}
