model Booking {
  id          String        @id @default(cuid())
  leadId      String
  buildingId  String
  bookedById  String
  status      BookingStatus @default(CONFIRMED)
  bookedAt    DateTime      @default(now())
  cancelledAt DateTime?

  lead     Lead     @relation(fields: [leadId], references: [id])
  building Building @relation(fields: [buildingId], references: [id])
  bookedBy User     @relation(fields: [bookedById], references: [id])

  @@index([leadId])
  @@index([buildingId, status])
}

model Building {
  id           String   @id @default(cuid())
  projectId    String
  name         String
  description  String?
  totalFloors  Int?
  contactName  String?
  contactPhone String?
  contactEmail String?
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  bookings Booking[]

  project Project         @relation(fields: [projectId], references: [id], onDelete: Cascade)
  units   PropertyUnit[]
  images  BuildingImage[]

  @@unique([projectId, name])

  building Building @relation(fields: [buildingId], references: [id], onDelete: Cascade)

  @@index([buildingId, sortOrder])
}

model Lead {
  id           String    @id @default(cuid())
  name         String
  email        String?
  phone        String
  source       String?
  stage        LeadStage @default(NEW)
  assignedToId String?
  createdById  String
  createdAt    DateTime  @default(now())
  updatedAt    DateTime  @updatedAt

  assignedTo User?      @relation("AssignedLeads", fields: [assignedToId], references: [id])
  createdBy  User       @relation("CreatedLeads", fields: [createdById], references: [id])
  notes      LeadNote[]
  followUps  FollowUp[]
  bookings   Booking[]

  @@index([assignedToId, stage])
  @@index([name])
  @@index([phone])
}

model LeadNote {
  id        String   @id @default(cuid())
  leadId    String
  authorId  String
  content   String
  createdAt DateTime @default(now())

  lead   Lead @relation(fields: [leadId], references: [id], onDelete: Cascade)
  author User @relation(fields: [authorId], references: [id])

  @@index([leadId, createdAt])
}

model PropertyUnit {
  id         String     @id @default(cuid())
  buildingId String
  unitNumber String
  type       UnitType
  price      Decimal    @db.Decimal(14, 2)
  bedrooms   Int?
  bathrooms  Int?
  area       Decimal?   @db.Decimal(10, 2)
  areaUnit   String?    @default("SQ_FT")
  status     UnitStatus @default(AVAILABLE)
  createdAt  DateTime   @default(now())
  updatedAt  DateTime   @updatedAt

  building Building @relation(fields: [buildingId], references: [id])

  images PropertyUnitImage[]

  @@unique([buildingId, unitNumber])
  @@index([status, type])
}

model User {

  createdLeads  Lead[]     @relation("CreatedLeads")
  leadNotes     LeadNote[]
  followUps     FollowUp[]
  bookings      Booking[]
